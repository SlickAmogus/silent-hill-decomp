# VSync (0 = off/uncapped, 1 = on, -1 = adaptive vsync)

# Target refresh rate in hz (0 = display default). Only applies in fullscreen mode.
# Set to your monitor's refresh rate (e.g. 60, 144, 240) combined with vsync = 1.
refresh_rate = 0

# Gameplay FPS cap (0 = uncapped, 30 = PSX-accurate, 60 = smooth).
# The PSX ran at 30fps (2 vblanks/frame). Set to 0 for uncapped or 60 for 60fps.
# Toggle at runtime with numpad 0. Game physics use real elapsed time so any cap is safe.
fps_cap = 30

# Disable aggressive PSX view culling (0 = original culling, 1 = render all objects)
# The original game aggressively culls objects based on view angle and position.
# Disabling culling prevents objects like fences and trees from popping in/out.
disable_culling = 1

# Preload all IPD chunks at map init (0 = PSX streaming, 1 = load all at once)
# The PSX streams 2-4 chunks from CD as the player moves. On PC, we can load
# all chunks for the entire map at once, eliminating streaming delays and
# ensuring all geometry is always available.
